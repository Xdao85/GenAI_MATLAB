What it contains

Simulation tab
A 3D ECEF view shows the Earth, a 24-satellite constellation (6 planes at 55° inclination), and each satellite's trail.
Satellites above the elevation mask are green and get a pseudorange line to the user. Hidden ones are grey and their lines are switched off.
The user walks on a 100 m circle. A local East/North plot compares the true path with the least-squares fixes.
A running plot shows the 3D position error, and a readout shows visible satellites, solver iterations, clock bias, and GDOP, PDOP, HDOP and VDOP.
Controls are time scale, noise σ, elevation mask, an iono-delay checkbox, and Start/Pause and Reset.
Equations tab
Eight groups of equations are rendered with text(..., 'Interpreter', 'latex'): pseudorange, geometric range and clock bias, linearisation, matrix form, least-squares update, DOP, orbit with Earth rotation, and elevation.
MATLAB Code tab
A list on the left selects a function. The panel on the right explains its role and which equation it implements, with the actual code snippet. Covered: vecnorm, randn, asin with the dot product, \, inv, sin/cos, LaTeX text, and set with drawnow limitrate.

Code structure

The file is divided into seven commented sections: parameters and constellation, GUI design, LaTeX integration, function explanations, visualization setup, callbacks and the simulation loop, and a slider helper. Three local functions do the GPS math: satPositions, solvePosition (a Gauss-Newton least-squares fix) and computeDOP.

Simplifications

The Earth is spherical and the orbits are circular.
The ionospheric delay is a simple uncorrected model, so unticking it removes that error source.
The user's motion is small compared with the 3D scale, which is why the local East/North plot is there.

To see the effects, raise the noise or the elevation mask and watch DOP and error grow. With a high mask the fix is lost when fewer than four satellites are visible.

    function onStartPause(~,~)
        % Start/pause toggle. The loop runs inside the callback; drawnow
        % lets the Pause/Reset buttons and sliders interrupt it.
        if running
            running = false;  set(btnRun,'String','Start');  return
        end
        running = true;  set(btnRun,'String','Pause');
        tPrev = tic;
        while running && isvalid(fig)
            dtReal = min(toc(tPrev),0.1);  tPrev = tic;
            stepSim(dtReal*get(slScale,'Value'));   % simulated seconds
            drawnow limitrate;
            pause(0.01);
        end
    end
 
    function onReset(~,~)
        running = false;  set(btnRun,'String','Start');
        t = 0;  xhat = zeros(4,1);  TR(:) = nan;
        hist = struct('t',[],'err',[],'E',[],'N',[]);
        stepSim(0);
    end
 
    function onClose(~,~)
        running = false;  delete(fig);     % stops the loop, closes window
    end
 
    function onDocSelect(src,~)
        set(edDoc,'String',docs{get(src,'Value'),2});

    function stepSim(dt)
        % ---- advance time and read UI settings ----
        t = t + dt;
        sigma = get(slNoise,'Value');  mask = get(slMask,'Value');
        useIono = get(chkIono,'Value');
 
        % ---- true satellite and user positions (ECEF) ----
        sat = satPositions(P,t);
        ang = 2*pi*t/P.Tpath;
        E = P.Rpath*cos(ang);  Nn = P.Rpath*sin(ang);
        ru = P.p0 + E*P.e + Nn*P.n;
 
        % ---- visibility from elevation angle (Eq. 8) ----
        los = sat - ru;  rng = vecnorm(los);
        el  = asin((P.u'*los)./rng);
        vis = rad2deg(el) >= mask;  nv = nnz(vis);
 
        % ---- pseudorange generation (Eq. 1) ----
        b = P.c*(P.clk0 + P.clkDrift*t);                  % clock bias [m]
        I = zeros(1,P.nSat);
        if useIono, I = 5*(1 + 16*(0.53 - el/pi).^3); end % simple iono model [m]
        rho = rng(vis) + b + I(vis) + sigma*randn(1,nv);
 
        % ---- least-squares fix and DOP (Eq. 3-6) ----
        fix = nv >= 4;
        if fix
            [xhat,G,it] = solvePosition(sat(:,vis),rho,xhat);
            dop = computeDOP(G,P.Renu);
            enu = P.Renu*(xhat(1:3) - P.p0);          % estimate in ENU about p0
            err = norm(xhat(1:3) - ru);
            hist.t = push(hist.t,t);   hist.err = push(hist.err,err);
            hist.E = push(hist.E,enu(1));  hist.N = push(hist.N,enu(2));
        end
 
        % ---- satellite trails (ring buffer) ----
        TR = [TR(2:end,:,:); reshape(sat'/1e3,1,P.nSat,3)];
 
        % ---- 3-D graphics update ----
        set(h.satVis,'XData',sat(1,vis)/1e3,'YData',sat(2,vis)/1e3,'ZData',sat(3,vis)/1e3);
        set(h.satHid,'XData',sat(1,~vis)/1e3,'YData',sat(2,~vis)/1e3,'ZData',sat(3,~vis)/1e3);
        set(h.user,'XData',ru(1)/1e3,'YData',ru(2)/1e3,'ZData',ru(3)/1e3);
        for k = 1:P.nSat
            set(h.trail(k),'XData',TR(:,k,1),'YData',TR(:,k,2),'ZData',TR(:,k,3));
            if vis(k)   % pseudorange line user -> satellite
                set(h.rng(k),'XData',[ru(1) sat(1,k)]/1e3,'YData',[ru(2) sat(2,k)]/1e3, ...
                    'ZData',[ru(3) sat(3,k)]/1e3,'Visible','on');
            else
                set(h.rng(k),'Visible','off');
            end
        end
        title(ax3d,sprintf('Constellation (ECEF)   t = %.0f s',t));
 
        % ---- ENU / error plots and read-out ----
        set(h.enuTrue,'XData',E,'YData',Nn);
        if fix
            set(h.estTrail,'XData',hist.E,'YData',hist.N);
            set(h.enuEst,'XData',enu(1),'YData',enu(2));
            set(h.err,'XData',hist.t,'YData',hist.err);
            set(readout,'String',{ ...
                sprintf('t = %7.1f s   Visible sats: %2d / %d   iterations: %d',t,nv,P.nSat,it)
                sprintf('3-D position error : %6.2f m',err)
                sprintf('Clock bias est/true: %.1f / %.1f m',xhat(4),b)
                sprintf('GDOP %.2f  PDOP %.2f  HDOP %.2f  VDOP %.2f',dop.G,dop.P,dop.H,dop.V)});
        else
            set(readout,'String',{sprintf('t = %.1f s   Visible: %d  -> NO FIX (need >= 4)',t,nv)});
        end
    end

%  LOCAL FUNCTIONS - GPS MATHEMATICS (simulation logic)

function s = satPositions(P,t)
% ECEF positions (3 x N) of circular-orbit satellites at time t. (Eq. 7)
u  = P.M0 + sqrt(P.mu/P.a^3)*t;                 % argument of latitude
cO = cos(P.raan); sO = sin(P.raan); ci = cos(P.inc); si = sin(P.inc);
x  = P.a*(cO.*cos(u) - sO.*ci.*sin(u));         % ECI coordinates
y  = P.a*(sO.*cos(u) + cO.*ci.*sin(u));
z  = P.a*si*sin(u);
th = P.we*t;                                    % Earth rotation angle
s  = [cos(th)*x + sin(th)*y; -sin(th)*x + cos(th)*y; z];   % ECI -> ECEF
end
 
function [x,G,it] = solvePosition(satV,rho,x0)
% Gauss-Newton least-squares solution for x = [xu;yu;zu;b]. (Eq. 3-5)
x = x0(:);
for it = 1:15
    d    = satV - x(1:3);                       % user -> satellite vectors
    r    = vecnorm(d);                          % predicted geometric ranges
    dRho = rho(:) - (r' + x(4));                % measured - predicted
    G    = [-(d./r)', ones(numel(r),1)];        % geometry matrix
    dx   = (G'*G) \ (G'*dRho);                  % normal equations
    x    = x + dx;                              % update estimate
    if norm(dx(1:3)) < 1e-4, break; end
end
end
 
function dop = computeDOP(G,Renu)
% Dilution of precision from the geometry matrix. (Eq. 6)
Q   = inv(G'*G);
Qe  = Renu*Q(1:3,1:3)*Renu';                    % position covariance in ENU
dop.G = sqrt(trace(Q));
dop.P = sqrt(trace(Q(1:3,1:3)));
dop.H = sqrt(Qe(1,1) + Qe(2,2));
dop.V = sqrt(Qe(3,3));
end
