The three tabs

Equations (LaTeX): the pseudorange equation, satellite orbit position, linearisation, least-squares update, GDOP and elevation are drawn with text(..., 'Interpreter','latex'). A notation key sits beside them.
MATLAB Functions: pick a function from the list to see an explanation and its source code. The code is read live from the file itself, so what you see is what runs. The functions are satelliteECI, computePseudoranges, solvePositionLS, computeDOP, elevationAngles, userTrajectory and stepSimulation.
Live Simulation: Start/Pause and Reset buttons start and stop the animation.
3D view: 24 satellites on 6 orbital planes move around the Earth. A moving user leaves a trail, with orange pseudorange lines to each visible satellite. Satellites below the elevation mask turn grey.
Position fix: a Gauss-Newton solver recomputes the user position and clock bias every frame. The 3D view marks the true user (star) and the estimate (red ×).
Error plots: an East/North error scatter shows the error in metres, since the error is too small to see in 3D. A second plot shows 3D error and GDOP over time.
Controls: sliders set the time step per frame, pseudorange noise, elevation mask and user speed. Checkboxes toggle the pseudorange lines and the orbits.

    function onRun(~,~)
        % Start/pause toggle; the while-loop IS the real-time animation.
        if running
            running = false; set(btnRun,'String','Start'); return;
        end
        running = true; set(btnRun,'String','Pause');
        while running && isvalid(fig)
            stepSimulation(round(get(slDt,'Value')));
            drawnow limitrate;
            pause(0.02);
        end
    end

    function stepSimulation(dt)
        % ---- read live GUI parameters
        sigma = get(slNoise,'Value')/1000;     % m -> km
        mask  = get(slMask,'Value');           % deg
        vel   = get(slSpeed,'Value');          % km/h
 
        % ---- advance time, user and satellites
        t = t + dt;
        [lat,lon] = userTrajectory(lat,lon,t,vel,dt,C.Re);
        th   = C.we * t;                                   % Earth rotation angle
        lamI = lon + th;                                   % inertial longitude
        uVec = C.Re*[cos(lat)*cos(lamI); cos(lat)*sin(lamI); sin(lat)];
        S    = satelliteECI(cn,C.a,C.inc,t);               % nSat x 3
 
        % ---- visibility
        el  = elevationAngles(S,uVec);
        vis = el >= mask;
        nv  = nnz(vis);
 
        % ---- pseudoranges and position solution
        bTrue = C.c*(1e-4 + 2e-9*t);                       % true clock bias [km]
        gdop = NaN; pdop = NaN; errM = NaN; fix = false; enu = [NaN NaN];
        xhat = xPrev;
        if nv >= 4
            rho = computePseudoranges(S(vis,:),uVec,bTrue,sigma);
            [xhat,H] = solvePositionLS(S(vis,:),rho,xPrev);
            xPrev = xhat; fix = true;
            [gdop,pdop] = computeDOP(H);
            e = xhat(1:3) - uVec;                          % ECI error vector
            up = uVec/norm(uVec);
            east = cross([0;0;1],up); east = east/norm(east);
            north = cross(up,east);
            enu  = 1000*[dot(e,east) dot(e,north)];        % m
            errM = 1000*norm(e);
            % ---- histories (capped length)
            trail   = [trail; uVec.'];             %#ok<AGROW>
            enuHist = [enuHist; enu];              %#ok<AGROW>
            tHist   = [tHist; t];                  %#ok<AGROW>
            errHist = [errHist; errM];             %#ok<AGROW>
            gdHist  = [gdHist; gdop];              %#ok<AGROW>
            if size(trail,1) > 3000
                trail = trail(end-2999:end,:); enuHist = enuHist(end-2999:end,:);
                tHist = tHist(end-2999:end); errHist = errHist(end-2999:end);
                gdHist = gdHist(end-2999:end);
            end
        end
 
        % ---- refresh visualization
        Sv = S(vis,:);  Sh = S(~vis,:);
        set(hSatVis,'XData',Sv(:,1),'YData',Sv(:,2),'ZData',Sv(:,3));
        set(hSatHid,'XData',Sh(:,1),'YData',Sh(:,2),'ZData',Sh(:,3));
        if get(chkLines,'Value') && nv > 0                 % pseudorange lines u -> s_i
            XL = [uVec(1)*ones(nv,1) Sv(:,1) nan(nv,1)].';
            YL = [uVec(2)*ones(nv,1) Sv(:,2) nan(nv,1)].';
            ZL = [uVec(3)*ones(nv,1) Sv(:,3) nan(nv,1)].';
            set(hLines,'XData',XL(:),'YData',YL(:),'ZData',ZL(:));
        else
            set(hLines,'XData',nan,'YData',nan,'ZData',nan);
        end
        set(hUser,'XData',uVec(1),'YData',uVec(2),'ZData',uVec(3));
        if fix
            set(hEst,'XData',xhat(1),'YData',xhat(2),'ZData',xhat(3));
        end
        if ~isempty(trail)
            set(hTrail,'XData',trail(:,1),'YData',trail(:,2),'ZData',trail(:,3));
        else
            set(hTrail,'XData',nan,'YData',nan,'ZData',nan);
        end
        if ~isempty(enuHist)
            set(hENU,'XData',enuHist(:,1),'YData',enuHist(:,2));
            set(hENUcur,'XData',enuHist(end,1),'YData',enuHist(end,2));
            set(hErr,'XData',tHist/60,'YData',errHist);
            set(hGd, 'XData',tHist/60,'YData',gdHist);
        else
            set(hENU,'XData',nan,'YData',nan); set(hENUcur,'XData',nan,'YData',nan);
            set(hErr,'XData',nan,'YData',nan); set(hGd,'XData',nan,'YData',nan);
        end
        title(ax3,sprintf('t = %.2f h   |   visible satellites: %d / %d',t/3600,nv,nSat), ...
            'Color','k');
 
        % ---- status panel
        if fix
            lines = { ...
              sprintf('Time        : %8.1f min',t/60)
              sprintf('Satellites  : %d visible',nv)
              sprintf('Lat / Lon   : %6.2f / %6.2f deg',rad2deg(lat),rad2deg(lon))

%% =====================================================================
%  LOCAL FUNCTIONS - SIMULATION LOGIC (shown verbatim in the "Functions" tab)
%  =====================================================================
function S = satelliteECI(cn,a,inc,t)
% Circular Keplerian orbit -> ECI position of every satellite (N x 3, km).
u  = cn.M0 + cn.n*t;                      % argument of latitude u_k = M0 + n t
xo = a*cos(u);  yo = a*sin(u);            % in-plane coordinates
X  = xo.*cos(cn.raan) - yo.*cos(inc).*sin(cn.raan);   % Rz(RAAN)*Rx(inc)
Y  = xo.*sin(cn.raan) + yo.*cos(inc).*cos(cn.raan);
Z  = yo.*sin(inc);
S  = [X Y Z];
end
 
function rho = computePseudoranges(S,u,b,sigma)
% rho_i = |s_i - u| + b + noise      (b = c*dt_u in km, sigma in km)
r   = vecnorm(S - u(:).', 2, 2);          % geometric range, N x 1
rho = r + b + sigma*randn(size(r));
end
 
function [x,H] = solvePositionLS(S,rho,x0)
% Gauss-Newton least squares for x = [x y z b]^T from N >= 4 pseudoranges.
x = x0(:);
for iter = 1:15
    d    = S - x(1:3).';                  % vectors from estimate to satellites
    r    = vecnorm(d,2,2);                % predicted ranges
    H    = [-d./r, ones(size(r))];        % geometry matrix
    dRho = rho - (r + x(4));              % pseudorange residual
    dx   = H \ dRho;                      % least-squares step (QR)
    x    = x + dx;                        % state update
    if norm(dx(1:3)) < 1e-7, break; end   % converged (< 0.1 mm)
end
d = S - x(1:3).';  r = vecnorm(d,2,2);    % H at the final estimate (for DOP)
H = [-d./r, ones(size(r))];
end
 
function [gdop,pdop] = computeDOP(H)
% Dilution of precision from the geometry matrix.
if rcond(H.'*H) < 1e-12
    gdop = NaN; pdop = NaN; return;       % degenerate geometry
end
Q    = inv(H.'*H);
gdop = sqrt(trace(Q));
pdop = sqrt(trace(Q(1:3,1:3)));
end
 
function el = elevationAngles(S,u)
% Elevation [deg] of every satellite as seen from user position u (ECI).
up = u(:)/norm(u);                        % local vertical
d  = S - u(:).';
el = asind((d*up) ./ vecnorm(d,2,2));
end
 
function [lat,lon] = userTrajectory(lat,lon,t,vel,dt,Re)
% Aircraft-like motion on the Earth surface with slowly varying heading.
hdg = 0.6 + 0.8*sin(t/2400);              % heading from north [rad]
ds  = vel/3600*dt;                        % distance travelled [km]
lat = lat + ds*cos(hdg)/Re;
lon = lon + ds*sin(hdg)/(Re*cos(lat));
lat = max(min(lat,deg2rad(80)),deg2rad(-80));
end

function src = getFunctionSource(name)
% Read this very file and return the source lines of function <name>
% (works for both local functions and the nested stepSimulation).
f = [mfilename('fullpath') '.m'];
if ~isfile(f)
    src = {'Source file not found - run gps_gui_simulator from a saved .m file.'};
    return;
end
L  = regexp(fileread(f),'\r?\n','split');
L  = regexprep(L,'\s+$','');
i0 = find(~cellfun(@isempty,regexp(L,['^\s*function\s+.*\<' name '\s*\('],'once')),1);
if isempty(i0), src = {['Function ' name ' not found.']}; return; end
ind = regexp(L{i0},'^\s*','match','once');
i1  = i0 + find(strcmp(L(i0+1:end),[ind 'end']),1);
src = L(i0:i1);
end
